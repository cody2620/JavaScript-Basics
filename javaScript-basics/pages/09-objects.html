객체

이름과 값의 쌍으로 정보를 저장하는 방법을 배웁니다.

📚 학습 목표

  • 객체 만들고 속성 접근하기

객체란?

객체는 키(key)와 값(value)의 쌍을 저장합니다. 현실의 "물건"을 코드로 표현하는 것 같습니다.

let person = {
  name: "철수",
  age: 20,
  city: "서울"
};

person.name;  // "철수"
person.age;   // 20
let book = {
  title: "JavaScript 입문",
  author: "김학습",
  pages: 300,
  published: 2024
};

console.log(book.title);      // JavaScript 입문
console.log(book.author);     // 김학습
console.log(book.pages);      // 300

객체 속성 접근하기

1. 점 표기법 (권장)

객체.속성
let car = {
  brand: "현대",
  model: "소나타",
  year: 2024
};

console.log(car.brand);  // 현대
console.log(car.model);  // 소나타

2. 대괄호 표기법

객체["속성"]
let person = {
  name: "철수",
  "favorite color": "파란색"  // 공백이 있으면 대괄호 필요
};

console.log(person["name"]);           // 철수
console.log(person["favorite color"]); // 파란색

객체 속성 추가/수정

let user = {name: "철수"};

// 속성 추가
user.age = 20;
user.email = "user@email.com";

// 속성 수정
user.name = "영희";

console.log(user);
// {name: "영희", age: 20, email: "user@email.com"}

메서드 — 객체 안의 함수

객체의 속성이 함수면 "메서드"라고 부릅니다.

let calculator = {
  x: 10,
  y: 5,
  add: function() {
    return this.x + this.y;
  },
  subtract: function() {
    return this.x - this.y;
  }
};

console.log(calculator.add());      // 15
console.log(calculator.subtract()); // 5

this 키워드

this는 현재 객체를 가리킵니다.

let person = {
  name: "철수",
  age: 20,
  greet: function() {
    console.log(`안녕하세요, 저는 ${this.name}이고 ${this.age}살입니다`);
  }
};

person.greet();  // 안녕하세요, 저는 철수이고 20살입니다

객체 구조 분해 (Destructuring)

객체의 속성을 변수로 쉽게 빼낼 수 있습니다.

let person = {name: "철수", age: 20, city: "서울"};

// 구조 분해
let {name, age} = person;

console.log(name);  // 철수
console.log(age);   // 20
let book = {
  title: "JavaScript 입문",
  author: "김학습",
  pages: 300
};

// 필요한 것만 빼내기
let {title, author} = book;

console.log(title);   // JavaScript 입문
console.log(author);  // 김학습

이름 바꾸기

let user = {name: "철수", email: "user@email.com"};

let {name: userName, email: userEmail} = user;

console.log(userName);   // 철수
console.log(userEmail);  // user@email.com

전개 연산자 (...)

객체를 복사하거나 합칠 때 사용합니다.

let person1 = {name: "철수", age: 20};
let person2 = {...person1, city: "서울"};

console.log(person2);  // {name: "철수", age: 20, city: "서울"}

// 속성 덮어쓰기
let updated = {...person1, age: 21};
console.log(updated);  // {name: "철수", age: 21}

JSON (JavaScript Object Notation)

객체를 문자열로 변환해서 저장하거나 전송할 때 사용합니다.

객체를 JSON 문자열로 (stringify)

let user = {
  name: "철수",
  age: 20,
  email: "user@email.com"
};

let jsonString = JSON.stringify(user);
console.log(jsonString);
// {"name":"철수","age":20,"email":"user@email.com"}

JSON 문자열을 객체로 (parse)

let jsonString = '{"name":"철수","age":20}';

let user = JSON.parse(jsonString);
console.log(user.name);  // 철수
console.log(user.age);   // 20

객체 메서드들

Object.keys — 키만 추출

let person = {name: "철수", age: 20, city: "서울"};

let keys = Object.keys(person);
console.log(keys);  // ["name", "age", "city"]

Object.values — 값만 추출

let person = {name: "철수", age: 20, city: "서울"};

let values = Object.values(person);
console.log(values);  // ["철수", 20, "서울"]

Object.entries — 키-값 쌍 추출

let person = {name: "철수", age: 20};

let entries = Object.entries(person);
console.log(entries);
// [["name", "철수"], ["age", 20]]

// for...of로 순회
for (let [key, value] of entries) {
  console.log(`${key}: ${value}`);
}

자주 하는 실수

⚠️ 실수 1: 따옴표 필요 없는 키에 따옴표 붙임

// OK (따옴표 없어도 됨)
let obj = {name: "철수", age: 20};

// OK (따옴표 있어도 됨)
let obj2 = {"name": "철수", "age": 20};

// 이런 경우엔 따옴표 필수
let obj3 = {"first-name": "철수"};  // 하이픈이 있으면 필수

⚠️ 실수 2: 메서드에서 this 잘못 사용

let obj = {
  name: "철수",
  greet: function() {
    console.log(this.name);  // ✅ 맞음
  }
};

// 일반 함수에서는 this가 다름
let func = obj.greet;
func();  // ❌ this가 obj가 아님

⚠️ 실수 3: undefined 속성 접근

let person = {name: "철수"};
console.log(person.age);     // undefined (존재하지 않는 속성)
console.log(person.email);   // undefined

해결: 미리 확인하세요.

if (person.age !== undefined) {
  console.log(person.age);
}

연습 문제

문제 1: 영화 정보 객체

다음 정보를 객체로 만들고 각 속성을 출력하세요:

제목: "라이언 일병 구하기", 감독: "스필버그", 년도: 1998

let movie = {
  title: "라이언 일병 구하기",
  director: "스필버그",
  year: 1998
};

console.log(movie.title);
console.log(movie.director);
console.log(movie.year);
💡 정답 확인

객체를 만들고 점 표기법으로 각 속성에 접근하면 됩니다.

문제 2: 메서드 만들기

정사각형 객체에 넓이를 계산하는 메서드를 추가하세요.

let square = {
  side: 5,
  area: function() {
    return this.side * this.side;
  }
};

console.log(square.area());  // 25
💡 정답 확인

함수를 메서드로 정의하고 this.side를 사용하면 됩니다.

문제 3: 구조 분해

다음 객체에서 이름과 이메일만 구조 분해로 추출하세요.

let user = {
  name: "철수",
  age: 20,
  email: "user@email.com",
  city: "서울"
};

let {name, email} = user;
console.log(name);    // 철수
console.log(email);   // user@email.com
💡 정답 확인

중괄호 안에 원하는 속성명을 넣으면 됩니다.
