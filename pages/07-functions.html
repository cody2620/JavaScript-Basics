함수란?

함수는 특정 일을 하는 코드 묶음입니다. 여러 번 재사용할 수 있습니다.

function sayHello() {
  console.log("안녕하세요!");
}

sayHello();  // 함수 호출
sayHello();  // 또 호출
function greet() {
  console.log("👋 반갑습니다!");
}

greet();  // 첫 번째 호출
greet();  // 두 번째 호출

함수 선언식 (Function Declaration)

가장 일반적인 함수 정의 방식입니다.

function 함수이름(매개변수1, 매개변수2) {
  // 함수 본문
  return 반환값;
}

매개변수 (Parameter)

함수가 받을 값들을 정의합니다.

function add(a, b) {
  console.log(`${a} + ${b} = ${a + b}`);
}

add(10, 5);   // 10 + 5 = 15
add(100, 200); // 100 + 200 = 300

반환값 (Return)

return으로 함수의 결과를 돌려줍니다.

function multiply(a, b) {
  return a * b;
}

let result = multiply(7, 3);
console.log(result);  // 21

console.log(multiply(4, 5));  // 20

여러 개의 return

function checkAge(age) {
  if (age >= 18) {
    return "성인";
  } else {
    return "미성년자";
  }
}

console.log(checkAge(25));  // 성인
console.log(checkAge(15));  // 미성년자

기본 매개변수 (Default Parameter)

매개변수를 전달하지 않으면 기본값을 사용합니다.

function greetUser(name = "손님") {
  console.log(`안녕하세요, ${name}님!`);
}

greetUser("철수");  // 안녕하세요, 철수님!
greetUser();        // 안녕하세요, 손님님!

함수 표현식 (Function Expression)

함수를 변수에 저장하는 방식입니다.

const 함수이름 = function(매개변수) {
  return 반환값;
};
const subtract = function(a, b) {
  return a - b;
};

console.log(subtract(10, 3));  // 7

화살표 함수 (Arrow Function) ⭐

더 간결한 함수 표현식입니다. 요즘 가장 많이 사용됩니다.

const 함수이름 = (매개변수) => {
  return 반환값;
};

// 또는 더 짧게
const 함수이름 = (매개변수) => 반환값;
// 일반 형태
const multiply = (a, b) => {
  return a * b;
};
console.log(multiply(3, 4));  // 12

// 짧은 형태 (한 줄)
const divide = (a, b) => a / b;
console.log(divide(12, 3));  // 4

매개변수가 1개면 괄호 생략 가능

const square = x => x * x;
console.log(square(5));  // 25

const greet = name => `반갑습니다, ${name}!`;
console.log(greet("영희"));  // 반갑습니다, 영희!

함수의 스코프

함수 안에서 정의한 변수는 함수 밖에서 사용할 수 없습니다.

function test() {
  let localVar = "함수 안";
  console.log(localVar);  // ✅ 가능
}

test();
console.log(localVar);  // ❌ 에러! ReferenceError

전역 vs 지역 변수

let globalVar = "전역";  // 전역 변수

function test() {
  let localVar = "지역";   // 지역 변수
  console.log(globalVar);  // ✅ 전역 변수는 함수에서 사용 가능
  console.log(localVar);   // ✅ 지역 변수는 함수 안에서만 사용 가능
}

test();
console.log(globalVar);  // ✅ 전역 변수는 항상 사용 가능
console.log(localVar);   // ❌ 지역 변수는 함수 밖에서 사용 불가

함수는 값입니다

함수를 다른 함수에 전달하거나 저장할 수 있습니다.

function execute(callback) {
  console.log("시작");
  callback();
  console.log("종료");
}

const sayHi = () => console.log("안녕!");

execute(sayHi);

자주 하는 실수

⚠️ 실수 1: return 빠뜨리기

function add(a, b) {
  a + b;  // return이 없으면 undefined 반환
}

let result = add(5, 3);
console.log(result);  // undefined

해결: return을 꼭 쓰세요.

function add(a, b) {
  return a + b;  // ✅
}

⚠️ 실수 2: 함수 호출 괄호 빠뜨리기

function sayHi() {
  console.log("안녕!");
}

sayHi;    // 함수 자체를 참조 (실행 안 됨)
sayHi();  // ✅ 함수 실행

⚠️ 실수 3: 스코프 혼동

function test() {
  let x = 10;
}

test();
console.log(x);  // ❌ 에러! 함수 밖에서 x를 사용할 수 없음

해결: 필요한 값은 return으로 반환하세요.

function test() {
  let x = 10;
  return x;
}

let result = test();
console.log(result);  // ✅ 10

연습 문제

문제 1: 인사 함수

이름을 받아서 "안녕하세요, [이름]님!"을 출력하는 함수를 만드세요.

// 함수를 작성하세요
function greet(name) {
  // 여기에 코드
}

greet("철수");  // 안녕하세요, 철수님!
💡 정답 보기 
function greet(name) {
  console.log(`안녕하세요, ${name}님!`);
}

greet("철수");  // 안녕하세요, 철수님!

문제 2: 두 수의 평균 구하기

두 수를 받아서 평균을 반환하는 함수를 만드세요.

const average = (a, b) => {
  // 화살표 함수로 작성하세요
  return /* 여기에 코드 */;
};

console.log(average(10, 20));  // 15
💡 정답 보기 
const average = (a, b) => (a + b) / 2;
console.log(average(10, 20));  // 15

문제 3: 절댓값 구하기

음수면 양수로, 양수면 그대로 반환하는 함수를 만드세요.

function abs(num) {
  if (num < 0) {
    return -num;
  } else {
    return num;
  }
}

console.log(abs(-5));   // 5
console.log(abs(10));   // 10
💡 정답 보기

위 코드가 정답입니다. 또는 삼항 연산자를 사용할 수도 있습니다:

function abs(num) {
  return num < 0 ? -num : num;
}
