배열이란?

배열은 여러 값을 하나의 변수에 저장할 때 사용합니다.

let fruits = ["딸기", "포도", "사과"];

// 각 요소는 인덱스(0부터 시작)로 접근
fruits[0];  // "딸기"
fruits[1];  // "포도"
fruits[2];  // "사과"
let numbers = [10, 20, 30, 40, 50];

console.log(numbers[0]);   // 10
console.log(numbers[2]);   // 30
console.log(numbers[4]);   // 50

// 배열의 길이
console.log(numbers.length);  // 5

배열 생성하기

대괄호 사용 (권장)

let empty = [];
let numbers = [1, 2, 3];
let mixed = [1, "text", true, null];

console.log(numbers);  // [1, 2, 3]
console.log(mixed);    // [1, "text", true, null]

Array 생성자 (거의 안 씀)

let arr = new Array(1, 2, 3);  // [1, 2, 3]

배열 메서드 — 배열 수정하기

push — 끝에 추가

let fruits = ["딸기", "포도"];
fruits.push("사과");      // 배열을 수정함
console.log(fruits);      // ["딸기", "포도", "사과"]

// 여러 개 추가도 가능
fruits.push("바나나", "오렌지");
console.log(fruits);

pop — 끝에서 제거

let numbers = [1, 2, 3, 4, 5];
let removed = numbers.pop();  // 마지막 요소 제거

console.log(removed);   // 5 (제거된 요소)
console.log(numbers);   // [1, 2, 3, 4]

unshift — 처음에 추가

let colors = ["빨강", "파랑"];
colors.unshift("노랑");  // 처음에 추가
console.log(colors);     // ["노랑", "빨강", "파랑"]

shift — 처음에서 제거

let colors = ["노랑", "빨강", "파랑"];
let first = colors.shift();  // 처음 요소 제거
console.log(first);  // "노랑"
console.log(colors); // ["빨강", "파랑"]

slice — 부분 추출 (복사)

배열.slice(시작, 끝)  // 끝 인덱스는 포함되지 않음
let numbers = [1, 2, 3, 4, 5];

console.log(numbers.slice(1, 3));   // [2, 3]
console.log(numbers.slice(2));      // [3, 4, 5] (끝 생략)
console.log(numbers);               // [1, 2, 3, 4, 5] (원본 변경 없음)

join — 문자열로 합치기

let words = ["Hello", "JavaScript", "World"];

console.log(words.join(" "));        // "Hello JavaScript World"
console.log(words.join("-"));        // "Hello-JavaScript-World"
console.log(words.join(""));         // "HelloJavaScriptWorld"

배열 순회 — forEach

배열의 각 요소에 대해 함수를 실행합니다.

let fruits = ["딸기", "포도", "사과"];

fruits.forEach(function(fruit) {
  console.log(fruit);
});
// 출력: 딸기, 포도, 사과

// 화살표 함수로 더 간단하게
fruits.forEach(fruit => console.log(fruit));

고차함수 (Higher-Order Functions) ⭐

배열의 각 요소를 변환하거나 필터링합니다.

map — 각 요소를 변환하기

배열의 모든 요소를 변환해서 새 배열을 만듭니다.

let numbers = [1, 2, 3, 4];

let doubled = numbers.map(num => num * 2);
console.log(doubled);  // [2, 4, 6, 8]

let squared = numbers.map(num => num ** 2);
console.log(squared);  // [1, 4, 9, 16]

// 원본은 변경되지 않음
console.log(numbers);  // [1, 2, 3, 4]

filter — 조건에 맞는 요소만 선택

let numbers = [1, 2, 3, 4, 5, 6];

// 짝수만 선택
let evens = numbers.filter(num => num % 2 === 0);
console.log(evens);  // [2, 4, 6]

// 3 이상인 수만 선택
let large = numbers.filter(num => num >= 3);
console.log(large);  // [3, 4, 5, 6]

find — 조건에 맞는 첫 요소 찾기

let users = [
  {name: "철수", age: 20},
  {name: "영희", age: 18},
  {name: "민준", age: 22}
];

let user = users.find(u => u.name === "영희");
console.log(user);  // {name: "영희", age: 18}

reduce — 모든 요소를 하나로 합치기

배열의 모든 요소를 합쳐서 하나의 값을 만듭니다.

let numbers = [1, 2, 3, 4, 5];

// 모두 더하기
let sum = numbers.reduce((total, num) => total + num, 0);
console.log(sum);  // 15

// 모두 곱하기
let product = numbers.reduce((total, num) => total * num, 1);
console.log(product);  // 120

includes — 요소 포함 여부 확인

let colors = ["빨강", "파랑", "초록"];

console.log(colors.includes("파랑"));   // true
console.log(colors.includes("노랑"));   // false

let numbers = [1, 2, 3, 4, 5];
console.log(numbers.includes(3));       // true

메서드 용도 원본 변경?
push 끝에 추가 O
pop 끝에서 제거 O
slice 부분 추출 X
map 각 요소 변환 X
filter 조건에 맞게 선택 X

자주 하는 실수

⚠️ 실수 1: 인덱스 0부터 시작하는 걸 잊기

let arr = ["a", "b", "c"];
arr[1];  // "b" (두 번째 요소)
arr[3];  // undefined (요소가 없음)

⚠️ 실수 2: slice는 원본을 변경하지 않음

let numbers = [1, 2, 3];
numbers.slice(0, 2);  // [1, 2]
console.log(numbers); // [1, 2, 3] (여전히 원본!)

// 올바른 사용
numbers = numbers.slice(0, 2);  // [1, 2]

연습 문제

문제 1: 배열의 합 구하기

reduce를 사용해서 배열의 모든 수의 합을 구하세요.

let numbers = [10, 20, 30, 40];
let sum = numbers.reduce((total, num) => {
  return total + num;
}, 0);
console.log(sum);  // 100
💡 정답 확인

10 + 20 + 30 + 40 = 100

문제 2: 이름만 뽑아내기

객체 배열에서 map을 사용해 이름만 뽑아내세요.

let users = [
  {name: "철수", age: 20},
  {name: "영희", age: 18}
];

let names = users.map(user => user.name);
console.log(names);  // ["철수", "영희"]
💡 정답 확인

map 함수로 각 객체의 name 속성을 추출하면 됩니다.

문제 3: 조건에 맞는 요소 필터링

10 이상의 수만 남기세요.

let numbers = [5, 15, 3, 20, 8];
let filtered = numbers.filter(num => num >= 10);
console.log(filtered);  // [15, 20]
💡 정답 확인

filter로 조건을 만족하는 요소만 선택하면 됩니다.
